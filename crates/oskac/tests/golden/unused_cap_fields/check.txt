cap struct Tools {
    out: Stdout,
    clock: Clock,
    net: Net,
}

fn stamp(tools: Tools) -> Int {
    match tools {
        Tools(out: _, clock: c, net: _) => c.now_millis()
    }
}

fn main(out: Stdout, clock: Clock, net: Net) -> Result<(), IoError> {
    let tools = Tools(out: out, clock: clock, net: net)
    let n = stamp(tools)
    tools.out.println("{n}")
}
